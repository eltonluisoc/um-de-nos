// ============================================ // LÓGICA PROFISSIONAL DE BUSCA // ============================================ const hoje = new Date(); const dataAtual = hoje.toISOString().split('T')[0]; // YYYY-MM-DD async function buscarSorteio() { try { await log('🔍 Iniciando busca automática...'); // 1. Conectar Firebase const app = initializeApp(firebaseConfig); const db = getFirestore(app); // 2. Buscar jogo ativo const jogosRef = collection(db, 'jogos'); const q = query(jogosRef, where('status', '==', 'aberto'), limit(1)); const querySnapshot = await getDocs(q); if (querySnapshot.empty) { await log('⚠️ Nenhum jogo ativo'); return; } const jogoDoc = querySnapshot.docs[0]; const jogoAtualId = jogoDoc.id; const jogoData = jogoDoc.data(); // 3. Buscar API da Quina const response = await fetch('https://loteriascaixa-api.herokuapp.com/api/quina/latest'); const dados = await response.json(); if (!dados || !dados.dezenas) { throw new Error('API sem dados'); } const numerosSorteados = dados.dezenas.map(Number); const concurso = dados.concurso; const dataSorteio = dados.data; // "DD/MM/YYYY" // 4. Converter data do sorteio para YYYY-MM-DD const partes = dataSorteio.split('/'); const dataSorteioFormatada = `${partes[2]}-${partes[1]}-${partes[0]}`; await log(`📅 Sorteio #${concurso} - Data: ${dataSorteioFormatada}`); await log(`📅 Hoje: ${dataAtual}`); // 5. VERIFICAR SE O SORTEIO É DE HOJE if (dataSorteioFormatada !== dataAtual) { await log(`⏸️ Sorteio #${concurso} não é de hoje. Aguardando...`); return; // Não importa sorteio de outro dia } // 6. VERIFICAR SE JÁ FOI IMPORTADO if (jogoData.ultimoConcursoImportado === concurso) { await log(`✅ Sorteio #${concurso} já foi importado.`); await log(`🛑 Busca finalizada para hoje.`); return; // Para a busca do dia } // 7. SORTEIO NOVO DETECTADO! await log(`🎉 SORTEIO #${concurso} DETECTADO!`); await log(`📊 Números: ${numerosSorteados.join(', ')}`); // ... (importar, salvar, atualizar acertos, enviar e-mail) await log(`✅ Sorteio #${concurso} importado com sucesso!`); await log(`🛑 Busca finalizada para hoje.`); } catch (error) { await log(`❌ ERRO: ${error.message}`); } }
